# Code to export Landstat information in Google Earth Engine

// ==========================
// 1. Define London area
// ==========================
var london = ee.Geometry.Rectangle([
  -0.5103, 51.2868,
   0.3340, 51.6919
]);

Map.centerObject(london, 10);

// ==========================
// 2. Load Landsat 8 L2
// ==========================
var image = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2')
  .filterBounds(london)
  .filterDate('2022-06-01', '2022-08-31')
  .filter(ee.Filter.lt('CLOUD_COVER', 10))
  .median()
  .clip(london);

// ==========================
// 3. Apply scale factors
// ==========================
var optical = image.select([
  'SR_B2', 'SR_B3', 'SR_B4',
  'SR_B5', 'SR_B6', 'SR_B7'
]).multiply(0.0000275).add(-0.2);

var thermal = image.select('ST_B10')
  .multiply(0.00341802)
  .add(149.0);

// ==========================
// 4. NDVI
// ==========================
var ndvi = optical.normalizedDifference([
  'SR_B5', // NIR
  'SR_B4'  // Red
]).rename('NDVI');

// ==========================
// 5. NDBI (NEW)
// ==========================
var ndbi = optical.normalizedDifference([
  'SR_B6', // SWIR
  'SR_B5'  // NIR
]).rename('NDBI');

// ==========================
// 6. LST in Celsius
// ==========================
var lst = thermal
  .subtract(273.15)
  .rename('LST');

// ==========================
// 7. Stack all variables
// ==========================
var ndvi = ndvi.float();
var ndbi = ndbi.float();
var lst = lst.float();

var output = ee.Image.cat([
  ndvi,
  ndbi,
  lst
]);

// ==========================
// 8. Export as GeoTIFF
// ==========================
Export.image.toDrive({
  image: output,
  description: 'London_LST_NDVI_NDBI',
  folder: 'GEE_Exports', // optional
  fileNamePrefix: 'London_LST_NDVI_NDBI',
  region: london,
  scale: 30,
  crs: 'EPSG:4326',
  maxPixels: 1e13,
  fileFormat: 'GeoTIFF'
});

// ==========================
// 9. Visualise layers
// ==========================
Map.addLayer(ndvi,
  {min: -1, max: 1, palette: ['brown', 'yellow', 'green']},
  'NDVI');

Map.addLayer(ndbi,
  {min: -1, max: 1, palette: ['blue', 'white', 'red']},
  'NDBI');

Map.addLayer(lst,
  {min: 15, max: 40, palette: ['blue', 'yellow', 'red']},
  'LST');
