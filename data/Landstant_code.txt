// ==========================
// 1. Define Greater London boundary
// ==========================
var london = ee.FeatureCollection('FAO/GAUL/2015/level2')
  .filter(ee.Filter.eq(
    'ADM0_NAME',
    'U.K. of Great Britain and Northern Ireland'
  ))
  .filter(ee.Filter.eq(
    'ADM2_NAME',
    'Greater London'
  ))
  .geometry();

Map.centerObject(london, 10);
Map.addLayer(london, {color: 'red'}, 'Greater London');


// ==========================
// 2. Load Landsat 8 L2
// ==========================
var image = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2')
  .filterBounds(london)
  .filterDate('2022-06-01', '2022-08-31')
  .filter(ee.Filter.lt('CLOUD_COVER', 10))
  .median()
  .clip(london);

// ==========================
// 3. Apply scale factors
// ==========================
var optical = image.select([
  'SR_B2', // Blue
  'SR_B3', // Green
  'SR_B4', // Red
  'SR_B5', // NIR
  'SR_B6', // SWIR1
  'SR_B7'  // SWIR2
]).multiply(0.0000275).add(-0.2);

var thermal = image.select('ST_B10')
  .multiply(0.00341802)
  .add(149.0);

// ==========================
// 4. NDVI
// ==========================
var ndvi = optical.normalizedDifference([
  'SR_B5',
  'SR_B4'
]).rename('NDVI');

// ==========================
// 5. NDBI
// ==========================
var ndbi = optical.normalizedDifference([
  'SR_B6',
  'SR_B5'
]).rename('NDBI');

// ==========================
// 6. MNDWI
// ==========================
var mndwi = optical.normalizedDifference([
  'SR_B3',
  'SR_B6'
]).rename('MNDWI');

// ==========================
// 7. Broadband Albedo
// Liang (2001) coefficients adapted for Landsat 8
// ==========================
var albedo = optical.expression(
  '0.356*B2 + 0.130*B4 + 0.373*B5 + 0.085*B6 + 0.072*B7 - 0.0018',
  {
    'B2': optical.select('SR_B2'),
    'B4': optical.select('SR_B4'),
    'B5': optical.select('SR_B5'),
    'B6': optical.select('SR_B6'),
    'B7': optical.select('SR_B7')
  }
).rename('ALBEDO');

// ==========================
// 8. Land Surface Temperature (°C)
// ==========================
var lst = thermal
  .subtract(273.15)
  .rename('LST');

// ==========================
// 9. Convert to float
// ==========================
optical = optical.float();
ndvi = ndvi.float();
ndbi = ndbi.float();
mndwi = mndwi.float();
albedo = albedo.float();
lst = lst.float();

// ==========================
// 10. Stack all variables
// ==========================
var output = ee.Image.cat([

  optical.select('SR_B2').rename('Blue'),
  optical.select('SR_B3').rename('Green'),
  optical.select('SR_B4').rename('Red'),
  optical.select('SR_B5').rename('NIR'),
  optical.select('SR_B6').rename('SWIR1'),
  optical.select('SR_B7').rename('SWIR2'),

  ndvi,
  ndbi,
  mndwi,
  albedo,
  lst

]);

// ==========================
// 11. Export GeoTIFF
// ==========================
Export.image.toDrive({
  image: output,
  description: 'London_Landsat_Features_2022',
  folder: 'GEE_Exports',
  fileNamePrefix: 'London_Landsat_Features_2022',
  region: london,
  scale: 30,
  crs: 'EPSG:4326',
  maxPixels: 1e13,
  fileFormat: 'GeoTIFF'
});

// ==========================
// 12. Visualisation
// ==========================

// True Colour
Map.addLayer(
  optical,
  {
    bands: ['SR_B4', 'SR_B3', 'SR_B2'],
    min: 0,
    max: 0.3
  },
  'True Colour'
);

// NDVI
Map.addLayer(
  ndvi,
  {
    min: -1,
    max: 1,
    palette: ['brown', 'yellow', 'green']
  },
  'NDVI'
);

// NDBI
Map.addLayer(
  ndbi,
  {
    min: -1,
    max: 1,
    palette: ['blue', 'white', 'red']
  },
  'NDBI'
);

// MNDWI
Map.addLayer(
  mndwi,
  {
    min: -1,
    max: 1,
    palette: ['brown', 'white', 'blue']
  },
  'MNDWI'
);

// Albedo
Map.addLayer(
  albedo,
  {
    min: 0.05,
    max: 0.40,
    palette: ['black', 'gray', 'white']
  },
  'Albedo'
);

// LST
Map.addLayer(
  lst,
  {
    min: 15,
    max: 40,
    palette: ['blue', 'yellow', 'red']
  },
  'LST'
);
